package com.xk.netprobe;

import android.content.Context;
import android.util.Log;

import java.io.BufferedReader;
import java.io.File;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.OutputStream;
import java.util.ArrayList;
import java.util.List;

public class NmapRunner {

    public interface Listener {
        void onLine(String line);
        void onExit(int code);
        void onError(String message);
    }

    private final Context ctx;
    private final PrivilegeManager.Level level;
    private Process process;
    private volatile boolean cancelled = false;

    public NmapRunner(Context ctx, PrivilegeManager.Level level) {
        this.ctx = ctx;
        this.level = level;
    }

    public void cancel() {
        cancelled = true;
        if (process != null) {
            try { process.destroy(); } catch (Exception ignored) {}
            process = null;
        }
    }

    public File ensureBinary() throws Exception {
        File out = new File(ctx.getFilesDir(), "nmap");
        if (out.exists() && out.length() > 0) {
            try { out.setExecutable(true, false); } catch (Exception ignored) {}
            return out;
        }
        try (InputStream in = ctx.getAssets().open("nmap");
             OutputStream os = new FileOutputStream(out)) {
            byte[] buf = new byte[65536];
            int n;
            while ((n = in.read(buf)) > 0) os.write(buf, 0, n);
        }
        out.setExecutable(true, false);
        try {
            Runtime.getRuntime().exec(new String[]{
                "chmod", "755", out.getAbsolutePath()
            }).waitFor();
        } catch (Exception ignored) {}
        return out;
    }

    public void run(File nmapBin, List<String> userArgs, Listener listener) {
        new Thread(() -> {
            try {
                List<String> args = adaptForPrivileges(userArgs);
                if (level == PrivilegeManager.Level.ROOT) {
                    runAsRoot(nmapBin, args, listener);
                } else {
                    runDirect(nmapBin, args, listener);
                }
            } catch (Exception e) {
                Log.e("NmapRunner", "error", e);
                listener.onError("Runner error: " + e.getMessage());
            }
        }).start();
    }

    private List<String> adaptForPrivileges(List<String> userArgs) {
        List<String> out = new ArrayList<>(userArgs);
        switch (level) {
            case APP:
                for (int i = out.size() - 1; i >= 0; i--) {
                    String a = out.get(i);
                    if (a.equals("-sS") || a.equals("-O") || a.equals("-sU")) {
                        out.remove(i);
                    }
                }
                if (!out.contains("--unprivileged")) out.add("--unprivileged");
                break;
            case SHELL:
                if (!out.contains("--unprivileged")) out.add("--unprivileged");
                for (int i = out.size() - 1; i >= 0; i--) {
                    String a = out.get(i);
                    if (a.equals("-sS") || a.equals("-O")) out.remove(i);
                }
                break;
            case ROOT:
                break;
        }
        if (!out.contains("-Pn")) out.add(0, "-Pn");
        if (!hasAnyT(out)) out.add("-T4");
        return out;
    }

    private boolean hasAnyT(List<String> args) {
        for (String a : args) if (a.startsWith("-T")) return true;
        return false;
    }

    private void runDirect(File nmapBin, List<String> args, Listener listener)
            throws Exception {
        List<String> cmd = new ArrayList<>();
        cmd.add(nmapBin.getAbsolutePath());
        cmd.addAll(args);
        ProcessBuilder pb = new ProcessBuilder(cmd);
        pb.redirectErrorStream(true);
        pb.environment().put("PATH", ctx.getFilesDir().getAbsolutePath()
                + ":/system/bin:/system/xbin");
        pb.directory(ctx.getFilesDir());
        process = pb.start();
        stream(process, listener);
    }

    private void runAsRoot(File nmapBin, List<String> args, Listener listener)
            throws Exception {
        StringBuilder cmd = new StringBuilder();
        cmd.append(nmapBin.getAbsolutePath());
        for (String a : args) cmd.append(' ').append(shellEscape(a));
        process = PrivilegeManager.execAsRoot(cmd.toString());
        stream(process, listener);
    }

    private void stream(Process p, Listener listener) {
        try {
            BufferedReader br = new BufferedReader(new InputStreamReader(p.getInputStream()));
            String line;
            while ((line = br.readLine()) != null) {
                if (cancelled) break;
                listener.onLine(line);
            }
            int code = p.waitFor();
            listener.onExit(code);
        } catch (Exception e) {
            listener.onError("stream error: " + e.getMessage());
        }
    }

    private String shellEscape(String s) {
        if (s.matches("[a-zA-Z0-9_/.:=-]+")) return s;
        return "'" + s.replace("'", "'\\''") + "'";
    }

    public static List<String> buildArgs(String target, String extra) {
        List<String> out = new ArrayList<>();
        if (extra != null && !extra.trim().isEmpty()) {
            for (String tok : extra.trim().split("\\s+")) {
                if (!tok.isEmpty()) out.add(tok);
            }
        }
        if (target != null && !target.trim().isEmpty()) out.add(target.trim());
        return out;
    }
}

package com.xk.netprobe;

import android.content.Context;
import android.content.pm.PackageManager;
import android.os.Build;
import android.util.Log;

import java.io.BufferedReader;
import java.io.DataOutputStream;
import java.io.InputStreamReader;

public class PrivilegeManager {

    public enum Level { APP, SHELL, ROOT }

    private static final String TAG = "PrivilegeManager";

    public static boolean testRealRoot() {
        Process p = null;
        try {
            p = Runtime.getRuntime().exec("su");
            DataOutputStream os = new DataOutputStream(p.getOutputStream());
            BufferedReader br = new BufferedReader(new InputStreamReader(p.getInputStream()));
            os.writeBytes("id\n");
            os.writeBytes("exit\n");
            os.flush();
            String line = br.readLine();
            if (line != null && line.contains("uid=0")) return true;
            p.waitFor();
        } catch (Exception e) {
            Log.d(TAG, "no su: " + e.getMessage());
        } finally {
            if (p != null) p.destroy();
        }
        return false;
    }

    public static boolean hasShizuku(Context ctx) {
        try {
            ctx.getPackageManager().getPackageInfo("moe.shizuku.privileged.api", 0);
            if (Build.VERSION.SDK_INT >= 23) {
                return ctx.checkSelfPermission("moe.shizuku.manager.permission.API_V23")
                        == PackageManager.PERMISSION_GRANTED;
            }
        } catch (Exception ignored) {}
        return false;
    }

    public static Level detect(Context ctx) {
        if (testRealRoot()) return Level.ROOT;
        if (hasShizuku(ctx)) return Level.SHELL;
        return Level.APP;
    }

    public static String describe(Level l) {
        switch (l) {
            case ROOT:  return "ROOT — full scans available";
            case SHELL: return "SHELL (Shizuku) — most scans";
            default:    return "APP — TCP connect scans";
        }
    }

    public static Process execAsRoot(String cmd) throws Exception {
        Process p = Runtime.getRuntime().exec("su");
        DataOutputStream os = new DataOutputStream(p.getOutputStream());
        os.writeBytes(cmd + "\n");
        os.writeBytes("exit\n");
        os.flush();
        return p;
    }
}

import android.os.Bundle;

public class TerminalActivity extends AppCompatActivity {

    private TextView tvCommand, tvOutput;
    private ScrollView scrollOutput;
    private ProgressBar progressBar;
    private NmapRunner runner;
    private final Handler ui = new Handler(Looper.getMainLooper());
    private StringBuilder buffer = new StringBuilder();
    private String commandShown = "";

    @Override
    protected void onCreate(Bundle s) {
        super.onCreate(s);
        setContentView(R.layout.activity_terminal);

        tvCommand = findViewById(R.id.tvCommand);
        tvOutput = findViewById(R.id.tvOutput);
        scrollOutput = findViewById(R.id.scrollOutput);
        progressBar = findViewById(R.id.progressBar);

        Button btnStop = findViewById(R.id.btnStop);
        Button btnClear = findViewById(R.id.btnClear);
        Button btnSave = findViewById(R.id.btnSave);

        PrivilegeManager.Level level = PrivilegeManager.detect(this);
        appendLine("[*] privilege: " + PrivilegeManager.describe(level));

        String target = getIntent().getStringExtra("target");
        String extra = getIntent().getStringExtra("extra");

        List<String> args = NmapRunner.buildArgs(target, extra);
        commandShown = "nmap " + joinArgs(args);
        tvCommand.setText("$ " + commandShown);

        btnStop.setOnClickListener(v -> {
            if (runner != null) runner.cancel();
            appendLine("[!] cancelled");
            progressBar.setVisibility(ProgressBar.GONE);
        });
        btnClear.setOnClickListener(v -> { buffer.setLength(0); tvOutput.setText(""); });
        btnSave.setOnClickListener(v -> saveOutput());

        runner = new NmapRunner(this, level);
        progressBar.setVisibility(ProgressBar.VISIBLE);

        new Thread(() -> {
            try {
                File bin = runner.ensureBinary();
                ui.post(() -> appendLine("[*] binary: " + bin.getAbsolutePath()));
                runner.run(bin, args, new NmapRunner.Listener() {
                    @Override public void onLine(String line) {
                        ui.post(() -> appendLine(line));
                    }
                    @Override public void onExit(int code) {
                        ui.post(() -> {
                            appendLine("");
                            appendLine("[*] exit code " + code);
                            progressBar.setVisibility(ProgressBar.GONE);
                        });
                    }
                    @Override public void onError(String message) {
                        ui.post(() -> {
                            appendLine("[!] " + message);
                            progressBar.setVisibility(ProgressBar.GONE);
                        });
                    }
                });
            } catch (Exception e) {
                ui.post(() -> {
                    appendLine("[!] nmap binary missing");
                    appendLine("[!] ضع nmap في app/src/main/assets/nmap");
                    appendLine("[!] أو: Termux → pkg install nmap");
                    progressBar.setVisibility(ProgressBar.GONE);
                });
            }
        }).start();
    }

    private void appendLine(String line) {
        buffer.append(line).append('\n');
        tvOutput.setText(buffer.toString());
        scrollOutput.post(() -> scrollOutput.fullScroll(ScrollView.FOCUS_DOWN));
    }

    private void saveOutput() {
        try {
            File dir = new File(getFilesDir(), "reports");
            if (!dir.exists() && !dir.mkdirs()) throw new RuntimeException("mkdir");
            String ts = new SimpleDateFormat("yyyyMMdd_HHmmss", Locale.US).format(new Date());
            File f = new File(dir, "nmap_" + ts + ".txt");
            try (FileOutputStream fos = new FileOutputStream(f)) {
                fos.write(("$ " + commandShown + "\n\n" + buffer.toString()).getBytes("UTF-8"));
            }
            Toast.makeText(this, "حُفظ: " + f.getName(), Toast.LENGTH_LONG).show();
        } catch (Exception e) {
            Toast.makeText(this, "فشل: " + e.getMessage(), Toast.LENGTH_LONG).show();
        }
    }

    private static String joinArgs(List<String> args) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < args.size(); i++) {
            if (i > 0) sb.append(' ');
            sb.append(args.get(i));
        }
        return sb.toString();
    }

    @Override
    protected void onDestroy() {
        super.onDestroy();
        if (runner != null) runner.cancel();
    }

import android.view.ViewGroup;
import android.widget.Button;
import android.widget.TextView;

import androidx.annotation.NonNull;
import androidx.appcompat.app.AppCompatActivity;
import androidx.recyclerview.widget.LinearLayoutManager;
import androidx.recyclerview.widget.RecyclerView;

import java.util.ArrayList;
import java.util.List;

public class WifiScannerActivity extends AppCompatActivity {

    private WifiManager wifi;
    private TextView tvStatus;
    private WifiAdapter adapter;
    private BroadcastReceiver receiver;

    @Override
    protected void onCreate(Bundle s) {
        super.onCreate(s);
        setContentView(R.layout.activity_wifi);

        wifi = (WifiManager) getApplicationContext().getSystemService(Context.WIFI_SERVICE);
        RecyclerView recycler = findViewById(R.id.recyclerWifi);
        tvStatus = findViewById(R.id.tvWifiStatus);
        Button btn = findViewById(R.id.btnScanWifi);

        adapter = new WifiAdapter();
        recycler.setLayoutManager(new LinearLayoutManager(this));
        recycler.setAdapter(adapter);

        receiver = new BroadcastReceiver() {
            @Override public void onReceive(Context c, Intent i) { showResults(); }
        };

        btn.setOnClickListener(v -> {
            try {
                registerReceiver(receiver, new IntentFilter(WifiManager.SCAN_RESULTS_AVAILABLE_ACTION));
                boolean started = wifi.startScan();
                tvStatus.setText(started ? "جاري الفحص..." : "فشل بدء الفحص");
            } catch (Exception e) {
                Toast.makeText(this, "خطأ: " + e.getMessage(), Toast.LENGTH_LONG).show();
            }
        });
    }

    private void showResults() {
        try {
            List<ScanResult> list = wifi.getScanResults();
            if (list == null) list = new ArrayList<>();
            adapter.setItems(list);
            tvStatus.setText("عدد الشبكات: " + list.size());
        } catch (Exception e) {
            tvStatus.setText("خطأ: " + e.getMessage());
        }
    }

    @Override
    protected void onDestroy() {
        super.onDestroy();
        try { unregisterReceiver(receiver); } catch (Exception ignored) {}
    }

    static class WifiAdapter extends RecyclerView.Adapter<WifiAdapter.VH> {
        private final List<ScanResult> items = new ArrayList<>();
        void setItems(List<ScanResult> l) { items.clear(); items.addAll(l); notifyDataSetChanged(); }
        @NonNull @Override public VH onCreateViewHolder(@NonNull ViewGroup p, int t) {
            return new VH(LayoutInflater.from(p.getContext()).inflate(R.layout.item_wifi, p, false));
        }
        @Override public void onBindViewHolder(@NonNull VH h, int i) {
            ScanResult r = items.get(i);
            h.ssid.setText(r.SSID == null || r.SSID.isEmpty() ? "(hidden)" : r.SSID);
            h.bssid.setText(r.BSSID);
            h.info.setText("Signal: " + r.level + " dBm  |  " + r.frequency + " MHz");
        }
        @Override public int getItemCount() { return items.size(); }
        static class VH extends RecyclerView.ViewHolder {
            TextView ssid, bssid, info;
            VH(@NonNull View v) {
                super(v);
                ssid = v.findViewById(R.id.tvWifiSsid);
                bssid = v.findViewById(R.id.tvWifiBssid);
                info = v.findViewById(R.id.tvWifiInfo);
            }
        }
    }
